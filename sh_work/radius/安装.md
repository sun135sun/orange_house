```


编辑: 
```
# sudo vim /etc/freeradius/3.0/mods-available/redis
ln -s /etc/freeradius/3.0/mods-available/redis /etc/freeradius/3.0/mods-enabled/ 
sudo vim /etc/freeradius/3.0/sites-enabled/default 

#将用户账号密码写入redis 再验证
if ("%{redis:hget user:%{User-Name} password}" != "") {
        update control {
            Cleartext-Password := "%{redis:hget user:%{User-Name} password}"
        }


    } else {
        reject
    }

post-auth {
#检查用户流量是否超额
check_redis_limit
}

sudo vim /usr/local/bin/check_redis_limit.sh

#!/bin/bash

USERNAME="$1"

# 从 Redis 获取 used 和 limit
USED=$(redis-cli HGET user:$USERNAME used)
LIMIT=$(redis-cli HGET user:$USERNAME limit)

# 如果 Redis 没有值，默认为 0
USED=${USED:-0}
LIMIT=${LIMIT:-0}

#echo "User: $USERNAME"
#echo "Used: $USED"
#echo "Limit: $LIMIT"

if [ "$USED" -ge "$LIMIT" ]; then
    #echo "Traffic limit exceeded"
    exit 1   # 返回非 0 表示超限
else
    echo "Traffic OK"
    exit 0   # 返回 0 表示没超限
fi


sudo vim /etc/freeradius/3.0/mods-enabled/check_redis_limit

exec check_redis_limit {
    wait = yes
    program = "/usr/local/bin/check_redis_limit.sh %{User-Name}"
}

# 建立流量计算脚本
```
sudo vim /usr/local/bin/update_redis_traffic.sh

#!/bin/bash

USER="$1"
IN="$2"
OUT="$3"

# 默认 Redis 使用 localhost
REDIS_CLI="/usr/bin/redis-cli"

# 使用 HINCRBY 增加流量
# 增加总使用量 = IN + OUT
TOTAL=$((IN + OUT))

$REDIS_CLI HINCRBY user:$USER used $TOTAL >/dev/null 2>&1

# 返回 0 表示成功
exit 0


sudo chmod +x /usr/local/bin/update_redis_traffic.sh


sudo vim /etc/freeradius/3.0/mods-enabled/update_redis_traffic

exec update_redis_traffic {
    wait = yes
    program = "/usr/local/bin/update_redis_traffic.sh %{User-Name} %{Acct-Input-Octets} %{Acct-Output-Octets}"
    input_pairs = request
    output = none
}


sudo vim /etc/freeradius/3.0/sites-enabled/default
accounting {
check_redis_limit
 if (Acct-Status-Type == "Interim-Update") {
        update_redis_traffic
    }
}
 

```
